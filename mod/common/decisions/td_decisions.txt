td_focus_north_africa = {
    is_shown = { this = c:SPA NOT = { has_variable = td_focus } NOT = { has_law = law_type:law_td_status_quo } }
    possible = {
        td_can_legislate_trigger = yes
        has_interest_marker_in_region = region_north_africa
        if = { limit = { has_law = law_type:law_td_historical_rights } bureaucracy >= 25 }
        if = { limit = { has_law = law_type:law_td_strategic_frontiers } bureaucracy >= 50 authority >= 25 }
        if = { limit = { has_law = law_type:law_td_imperial_expansion } bureaucracy >= 75 authority >= 50 }
    }
    when_taken = {
        set_variable = td_focus
        if = {
            limit = { has_law = law_type:law_td_historical_rights }
            add_modifier = td_focus_historical_rights
            set_variable = { name = td_focus_wait months = 6 }
        }
        if = {
            limit = { has_law = law_type:law_td_strategic_frontiers }
            add_modifier = td_focus_strategic_frontiers
            set_variable = { name = td_focus_wait months = 6 }
        }
        if = {
            limit = { has_law = law_type:law_td_imperial_expansion }
            add_modifier = td_focus_imperial_expansion
            set_variable = { name = td_focus_wait months = 12 }
        }
    }
    ai_chance = { value = 0 }
}

}
