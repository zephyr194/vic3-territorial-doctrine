on_game_started = { on_actions = { td_initialize_world } }
td_initialize_world = {
    effect = {
        set_global_variable = td_mod_present
        if = { limit = { td_feature_enabled_trigger = yes }
            if = { limit = { NOT = { has_global_variable = td_original_registry_ready } }
                every_state = { td_register_original_territory_effect = yes }
                set_global_variable = td_original_registry_ready
            }
            every_country = { td_initialize_effect = yes td_refresh_country_civic_effect = yes }
            every_state = { td_update_state_owner_effect = yes }
        }
        else = { every_country = { td_disabled_cleanup_effect = yes } }
    }
}
on_state_created = { on_actions = { td_new_state_created } }
td_new_state_created = {
    effect = {
        if = { limit = { td_feature_enabled_trigger = yes }
            set_variable = { name = td_acquired_owner value = owner }
            if = {
                limit = { owner = { has_variable = td_initialized_v4 } is_treaty_port = no NOT = { is_homeland_of_country_cultures = owner } }
                set_state_type = unincorporated
            }
            td_update_state_owner_effect = yes
        }
    }
}
on_colony_created = { on_actions = { td_colony_created } }
td_colony_created = {
    effect = {
        if = { limit = { td_feature_enabled_trigger = yes }
            set_variable = td_colonial_origin
            set_variable = { name = td_acquired_owner value = owner }
            td_update_state_owner_effect = yes
        }
    }
}
on_state_owner_change = { on_actions = { td_state_changed } }
td_state_changed = { effect = { td_update_state_owner_effect = yes } }
on_monthly_pulse_country = { on_actions = { td_country_monthly } }
td_country_monthly = {
    effect = {
        if = { limit = { td_feature_enabled_trigger = yes }
            td_initialize_effect = yes
            every_scope_state = { td_update_state_owner_effect = yes }
            td_refresh_success_credit_effect = yes
            # AI chooses an eligible target and negotiates through the same
            # funded commitments; no additional focus-maturity wait applies.
            if = { limit = { is_player = no td_can_start_bill_trigger = yes NOT = { has_law = law_type:law_td_status_quo } any_state = { td_claim_candidate_trigger = yes } }
                td_select_claim_effect = yes
                if = { limit = { td_claim_target_valid_trigger = yes OR = { var:td_target = { td_colonial_target_trigger = no } td_colonial_expansion_pull >= 0 } }
                    if = { limit = { td_security_reason_valid_trigger = yes } set_variable = { name = td_reason value = 1 } }
                    else_if = { limit = { td_legal_reason_valid_trigger = yes } set_variable = { name = td_reason value = 2 } }
                    else_if = { limit = { td_colonial_reason_valid_trigger = yes } set_variable = { name = td_reason value = 6 } }
                    else = { set_variable = { name = td_reason value = 7 } }
                    set_variable = { name = td_bill value = 1 }
                    td_start_bill_effect = yes
                }
            }
            if = { limit = { is_player = no td_can_start_bill_trigger = yes any_scope_state = { td_needs_admission_trigger = yes } }
                td_select_owned_effect = yes
                if = { limit = { OR = { var:td_target = { td_colonial_target_trigger = no } td_colonial_admission_pull >= 0 } }
                    set_variable = { name = td_reason value = 4 }
                    set_variable = { name = td_bill value = 2 }
                    td_start_bill_effect = yes
                }
            }
            # Previously admitted states that lacked budget may start later.
            if = { limit = { is_player = no td_can_select_target_trigger = yes any_scope_state = { td_non_homeland_state_trigger = yes has_variable = td_admitted_owner var:td_admitted_owner = owner } }
                td_select_admitted_effect = yes
                if = { limit = { exists = var:td_target td_fast_ready_trigger = yes } td_start_fast_effect = yes }
            }
            td_allocations_monthly_effect = yes
            td_pledges_monthly_effect = yes
            td_refresh_country_civic_effect = yes
            if = { limit = { OR = { has_variable = td_fast has_variable = td_regular } }
                if = { limit = { OR = { NOT = { exists = var:td_project_target } var:td_project_target = { NOT = { owner = root } } var:td_project_target = { is_incorporated = yes } } }
                    td_end_project_effect = yes
                }
            }
        }
        else = { td_disabled_cleanup_effect = yes }
    }
}
