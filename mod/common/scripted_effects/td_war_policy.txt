# COUNTRY ROOT. Called by country event td.27, with a saved war scope.
# Only the attacking warleader is assessed. Defense/return_state/reparations
# supply no expansion targets. Snapshot failures cannot be erased mid-war.
td_snapshot_war_mandate_effect = {
    if = {
        limit = {
            td_feature_enabled_trigger = yes
            exists = scope:td_mandate_war
            scope:td_mandate_war = {
                is_war_participant = root
                OR = {
                    AND = { NOT = { has_variable = td_policy_checked } attacker_warleader = root }
                    AND = { has_variable = td_policy_checked var:td_policy_country = root }
                }
            }
        }
        scope:td_mandate_war = {
            if = { limit = { NOT = { has_variable = td_policy_checked } }
                set_variable = td_policy_checked
                set_variable = { name = td_policy_country value = root }
                set_variable = { name = td_extra_exhaustion value = 0 }
            }
        }
        if = {
            limit = {
                any_state = { td_war_expansion_target_trigger = yes }
                OR = {
                    NOT = { td_expansion_doctrine_trigger = yes }
                    any_state = { td_war_expansion_target_trigger = yes NOT = { td_state_has_mandate_trigger = yes } }
                }
            }
            scope:td_mandate_war = { set_variable = td_unauthorized_war }
        }
    }
}
# COUNTRY monthly. Counters live on each war; monthly events do not schedule
# additional monthly events, and the saved war disappears after peace.
td_war_monthly_effect = {
    # Authorization remains diagnostic; warfare uses vanilla losses only.
    set_variable = { name = td_penalized_wars value = 0 }
}
