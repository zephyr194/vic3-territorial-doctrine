# STATE scope. Ownership changes invalidate approval without changing homelands.
td_update_state_owner_effect = {
    if = {
        limit = { td_feature_enabled_trigger = yes }
        if = {
            limit = { has_variable = td_recorded_owner NOT = { var:td_recorded_owner = owner } }
            if = { limit = { exists = var:td_recorded_owner } var:td_recorded_owner = {
                td_revoke_success_on_prev_state_effect = yes
                td_store_power_on_prev_state_effect = yes
            } }
            set_variable = { name = td_source_administration value = var:td_live_administration }
            set_variable = { name = td_source_power value = var:td_live_power }
            remove_variable = td_admitted_owner
            remove_variable = td_admission_reason
            remove_variable = td_integration_project_owner
            remove_modifier = td_admission_ratified
            remove_modifier = td_annual_integration
            set_variable = { name = td_acquired_owner value = owner }
            if = {
                limit = { is_treaty_port = no NOT = { is_homeland_of_country_cultures = owner } }
                set_state_type = unincorporated
            }
        }
        if = {
            limit = { NOT = { has_variable = td_source_administration } }
            # Existing colonies / freshly split states have no reliable former-owner record.
            set_variable = { name = td_source_administration value = 1 }
            set_variable = { name = td_source_power value = 0 }
        }
        owner = { td_store_power_on_prev_state_effect = yes }
        set_variable = { name = td_recorded_owner value = owner }
        if = {
            limit = { td_non_homeland_state_trigger = yes }
            if = {
                limit = { NOT = { AND = { has_variable = td_admitted_owner var:td_admitted_owner = owner has_variable = td_integration_project_owner var:td_integration_project_owner = owner } } }
                if = { limit = { NOT = { has_modifier = td_requires_admission } } add_modifier = td_requires_admission }
            }
            else = { remove_modifier = td_requires_admission }
        }
        else = { remove_modifier = td_requires_admission }
        td_refresh_state_development_effect = yes
        td_process_acquisition_effect = yes
    }
}

# COUNTRY scope; selection itself never passes legislation.
td_pick_owned_effect = {
    ordered_scope_state = {
        limit = {
            td_needs_admission_trigger = yes
            NOT = { root = { is_target_in_variable_list = { name = td_seen_owned_states target = prev } } }
        }
        order_by = state_population
        max = 1
        root = {
            set_variable = { name = td_target value = prev }
            add_to_variable_list = { name = td_seen_owned_states target = prev }
        }
    }
}
td_select_owned_effect = {
    if = {
        limit = { td_can_start_bill_trigger = yes }
        remove_variable = td_target
        td_pick_owned_effect = yes
        if = { limit = { NOT = { has_variable = td_target } } clear_variable_list = td_seen_owned_states td_pick_owned_effect = yes }
        if = { limit = { exists = var:td_target } set_variable = { name = td_pending_bill value = 2 } }
        else = { remove_variable = td_pending_bill }
        set_variable = { name = td_reason value = 0 }
        td_refresh_target_effect = yes
    }
}

    }

        limit = { td_can_select_target_trigger = yes }

        td_pick_admitted_effect = yes

            limit = { exists = var:td_target }
            set_variable = { name = td_reason value = var:td_target.var:td_admission_reason }
        }
        td_refresh_target_effect = yes
    }
}

# COUNTRY: a target may disappear while a browsing window is open. Never open
# a dynamic state title or cost preview without an existing selected state.
td_open_target_menu_effect = {
    if = { limit = { exists = var:td_target } trigger_event = td.20 }
    else = { trigger_event = td.31 }
}
td_open_integration_menu_effect = {
    if = { limit = { exists = var:td_target } trigger_event = td.22 }
    else = { trigger_event = td.31 }
}
td_designate_focus_effect = {
    if = { limit = { td_can_designate_focus_trigger = yes }
        var:td_target.region = { save_scope_as = td_new_focus_region }
        save_scope_value_as = { name = td_focus_run value = var:td_focus_epoch }
        add_to_variable_list = { name = td_focus_regions target = scope:td_new_focus_region }
        change_variable = { name = td_focus_count add = 1 }
        remove_modifier = td_focus_historical_rights
        remove_modifier = td_focus_strategic_frontiers
        remove_modifier = td_focus_imperial_expansion
        if = { limit = { has_law = law_type:law_td_historical_rights }
            add_modifier = { name = td_focus_historical_rights multiplier = var:td_focus_count }
            trigger_event = { id = td.25 months = 6 }
        }
        if = { limit = { has_law = law_type:law_td_strategic_frontiers }
            add_modifier = { name = td_focus_strategic_frontiers multiplier = var:td_focus_count }
            trigger_event = { id = td.25 months = 6 }
        }
        if = { limit = { has_law = law_type:law_td_imperial_expansion }
            add_modifier = { name = td_focus_imperial_expansion multiplier = var:td_focus_count }
            trigger_event = { id = td.25 months = 12 }
        }
    }
}
