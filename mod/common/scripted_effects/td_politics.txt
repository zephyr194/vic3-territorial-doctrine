td_start_bill_effect = {
    if = {
        limit = { td_feature_enabled_trigger = yes }
        if = { limit = { NOT = { has_variable = td_bill_generation } } set_variable = { name = td_bill_generation value = 0 } }
        change_variable = { name = td_bill_generation add = 1 }
        set_variable = { name = td_bill_target value = var:td_target }
        save_scope_value_as = { name = td_bill_run value = var:td_bill_generation }
        td_refresh_target_effect = yes
        set_variable = { name = td_round_epoch value = 0 }
        save_scope_value_as = { name = td_round_run value = var:td_round_epoch }
        set_variable = { name = td_successes value = 0 }
        set_variable = { name = td_failures value = 0 }
        td_calculate_politics_effect = yes
        trigger_event = { id = td.1 days = td_bill_round_days }
    }
}

td_open_negotiation_effect = {
    if = { limit = { has_variable = td_bill has_variable = td_waiting_round }
        if = { limit = { exists = var:td_bill_target }
            set_variable = { name = td_target value = var:td_bill_target }
            if = { limit = { td_bill_target_valid_trigger = yes td_can_legislate_trigger = yes }
                save_scope_value_as = { name = td_bill_run value = var:td_bill_generation }
                save_scope_value_as = { name = td_round_run value = var:td_round_epoch }
                var:td_bill_target = { save_scope_as = td_negotiation_target }
                trigger_event = td.32
            }
            else = { td_cancel_bill_effect = yes trigger_event = td.14 }
        }
        else = { td_cancel_bill_effect = yes trigger_event = td.14 }
    }
}
td_resolve_round_effect = {
    if = { limit = { td_active_negotiation_trigger = yes }
        set_variable = { name = td_target value = var:td_bill_target }
        if = { limit = { td_bill_target_valid_trigger = no }
            td_cancel_bill_effect = yes
            trigger_event = td.14
        }
        else = {
            remove_variable = td_waiting_round
            td_calculate_politics_effect = yes
            random_list = {
                1 = { modifier = { multiply = var:td_chance } change_variable = { name = td_successes add = 1 } }
                1 = { modifier = { multiply = td_failure_weight } change_variable = { name = td_failures add = 1 } }
            }
            if = { limit = { var:td_successes >= td_bill_successes } td_pass_bill_effect = yes }
            else_if = { limit = { var:td_failures >= td_bill_failures } td_cancel_bill_effect = yes trigger_event = td.13 }
            else = { trigger_event = { id = td.1 days = td_bill_round_days } }
        }
    }
}

td_pass_bill_effect = {
    if = {
        limit = { td_feature_enabled_trigger = yes }
        if = { limit = { exists = var:td_bill_target } set_variable = { name = td_target value = var:td_bill_target } }
        set_variable = { name = td_completed_bill value = var:td_bill }
        td_ratify_pledges_effect = yes
        if = {
            limit = { var:td_bill = 1 }
            set_variable = { name = td_claim_infamy_snapshot value = td_claim_infamy }
            var:td_target.state_region = { add_claim = root }
            change_infamy = var:td_claim_infamy_snapshot
            if = { limit = { NOT = { is_target_in_variable_list = { name = td_approved_claim_regions target = var:td_target.state_region } } } add_to_variable_list = { name = td_approved_claim_regions target = var:td_target.state_region } }
            var:td_target = { save_scope_as = td_notice_target }
            trigger_event = td.11
        }
        else_if = {
            limit = { var:td_bill = 2 }
            var:td_target = {
                set_variable = { name = td_admitted_owner value = owner }
                set_variable = { name = td_admission_reason value = root.var:td_reason }
                add_modifier = td_admission_ratified
            }
            set_variable = { name = td_last_amendment_target value = var:td_target }
            var:td_target = { save_scope_as = td_notice_target }
            trigger_event = td.12
        }
        else = {
            var:td_target.state_region = { remove_claim = root }
            add_to_variable_list = { name = td_renounced_regions target = var:td_target.state_region }
            var:td_target = { save_scope_as = td_notice_target }
            trigger_event = td.19
        }
        td_clear_bill_effect = yes
        if = { limit = { var:td_completed_bill = 2 td_fast_ready_trigger = yes } td_start_fast_effect = yes }
        remove_variable = td_completed_bill
    }
}
