td_snapshot_navy_baseline_effect = {
    if = { limit = { exists = var:td_navy_target }
        set_variable = { name = td_navy_baseline value = 0 }
        set_variable = { name = td_navy_probe value = 1 }
        while = {
            limit = { var:td_navy_target = { state_has_building_type_levels = { target = bt:building_naval_base value >= root.var:td_navy_probe } } }
            change_variable = { name = td_navy_probe multiply = 2 }
        }
        while = {
            limit = { var:td_navy_probe > 1 }
            change_variable = { name = td_navy_probe divide = 2 }
            set_variable = { name = td_navy_candidate value = { value = var:td_navy_baseline add = var:td_navy_probe } }
            if = { limit = { var:td_navy_target = { state_has_building_type_levels = { target = bt:building_naval_base value >= root.var:td_navy_candidate } } }
                set_variable = { name = td_navy_baseline value = var:td_navy_candidate }
            }
        }
        remove_variable = td_navy_probe
        remove_variable = td_navy_candidate
    }
}
td_clear_navy_pledge_effect = {
    if = { limit = { has_variable = td_bill has_variable = td_bill_pledge_navy var:td_navy_bill_generation = var:td_bill_generation }
        remove_variable = td_bill_pledge_navy
        change_variable = { name = td_bill_allocation_count subtract = 1 }
    }
    remove_modifier = td_navy_planning
    remove_variable = td_navy_target
    remove_variable = td_navy_region
    remove_variable = td_navy_baseline
    remove_variable = td_navy_deadline
    remove_variable = td_navy_acquisition_deadline
    remove_variable = td_navy_started
    remove_variable = td_navy_ratified
    remove_variable = td_navy_owned_at_sign
    remove_variable = td_navy_bill_generation
    remove_variable = td_navy_survey_paid
    remove_variable = td_pending_navy
}
td_sign_navy_pledge_effect = {
    if = { limit = { td_can_pledge_navy_trigger = yes }
        set_variable = { name = td_target value = var:td_bill_target }
        if = { limit = { td_bill_target_valid_trigger = yes }
            set_variable = { name = td_navy_survey_paid value = td_pledge_survey_cost }
            add_treasury = { value = var:td_navy_survey_paid multiply = -1 }
            add_modifier = td_navy_planning
            set_variable = td_pending_navy
            set_variable = td_bill_pledge_navy
            change_variable = { name = td_bill_allocation_count add = 1 }
            set_variable = { name = td_navy_bill_generation value = var:td_bill_generation }
            set_variable = { name = td_navy_region value = var:td_bill_target.state_region }
            if = { limit = { var:td_bill_target = { owner = root } }
                set_variable = td_navy_owned_at_sign
                set_variable = { name = td_navy_target value = var:td_bill_target }
                td_snapshot_navy_baseline_effect = yes
            }
            td_calculate_politics_effect = yes
        }
    }
}

td_pledges_monthly_effect = {
    if = { limit = { td_feature_enabled_trigger = yes }
        if = { limit = { has_variable = td_pending_navy has_variable = td_navy_ratified }
            if = { limit = { NOT = { has_variable = td_navy_started } }
                if = { limit = { NOT = { has_variable = td_navy_acquisition_deadline } }
                    td_clear_navy_pledge_effect = yes
                }
                else = {
                    ordered_scope_state = {
                        limit = { state_region = root.var:td_navy_region is_treaty_port = no }
                        order_by = state_population
                        max = 1
                        root = {
                            set_variable = { name = td_navy_target value = prev }
                            td_snapshot_navy_baseline_effect = yes
                            set_variable = td_navy_started
                            set_variable = { name = td_navy_deadline years = 3 }
                            remove_variable = td_navy_acquisition_deadline
                        }
                    }
                }
            }
            if = { limit = { has_variable = td_pending_navy has_variable = td_navy_started }
                if = { limit = { NOT = { has_variable = td_navy_deadline } }
                    ig:ig_armed_forces = {
                        remove_modifier = td_pledge_broken
                        add_modifier = { name = td_pledge_broken years = 5 }
                        set_variable = { name = td_broken_pledge years = 5 }
                    }
                    td_clear_navy_pledge_effect = yes
                    trigger_event = td.18
                }
                else_if = { limit = { exists = var:td_navy_target var:td_navy_target = {
                    owner = root
                    is_treaty_port = no
                    state_has_building_type_levels = { target = bt:building_naval_base value > root.var:td_navy_baseline }
                } }
                    ig:ig_armed_forces = { remove_modifier = td_pledge_fulfilled add_modifier = { name = td_pledge_fulfilled years = 3 } }
                    td_clear_navy_pledge_effect = yes
                    trigger_event = td.17

                        limit = { state_region = root.var:td_railway_region is_treaty_port = no }

                        max = 1

                            set_variable = { name = td_railway_target value = prev }

                            set_variable = td_railway_started
                            set_variable = { name = td_railway_deadline years = 3 }
                            remove_variable = td_railway_acquisition_deadline
                        }
                    }
                }
            }
            if = { limit = { has_variable = td_pending_railway has_variable = td_railway_started }
                if = { limit = { NOT = { has_variable = td_railway_deadline } }
                    ig:ig_industrialists = {
                        remove_modifier = td_pledge_broken
                        add_modifier = { name = td_pledge_broken years = 5 }
                        set_variable = { name = td_broken_pledge years = 5 }
