# Delayed country event. It never schedules another round after terminal cleanup.
td.1 = {
    type = country_event
    hidden = yes
    immediate = {
        if = {
            limit = { td_feature_enabled_trigger = yes has_variable = td_bill scope:td_bill_run = var:td_bill_generation scope:td_round_run = var:td_round_epoch NOT = { has_variable = td_waiting_round } }
            set_variable = { name = td_target value = var:td_bill_target }
            if = {
                limit = { td_bill_target_valid_trigger = no td_can_legislate_trigger = yes }
                td_cancel_bill_effect = yes
                trigger_event = td.14
            }
            else_if = {
                limit = { td_can_legislate_trigger = no }
                td_cancel_bill_effect = yes
                trigger_event = td.14
            }
            else = {
                td_calculate_politics_effect = yes
                change_variable = { name = td_round_epoch add = 1 }
                save_scope_value_as = { name = td_round_run value = var:td_round_epoch }
                set_variable = td_waiting_round
                if = { limit = { is_player = yes } td_open_negotiation_effect = yes }
                else = {
                    if = { limit = { td_can_pledge_navy_trigger = yes } td_sign_navy_pledge_effect = yes }
                    if = { limit = { td_can_pledge_railway_trigger = yes } td_sign_railway_pledge_effect = yes }
                    td_resolve_round_effect = yes
                }
            }
        }
    }
}
