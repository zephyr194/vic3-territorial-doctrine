namespace = td

# Delayed country event. It never schedules another round after terminal cleanup.
td.1 = {
    type = country_event
    hidden = yes
    immediate = {
        if = {
            limit = { has_variable = td_bill }
            if = {
                limit = { td_bill_target_valid_trigger = no td_can_legislate_trigger = yes }
                td_cancel_bill_effect = yes
                trigger_event = td.14
            }
            else_if = {
                limit = { td_can_legislate_trigger = no }
                td_cancel_bill_effect = yes
                trigger_event = td.14
            }
            else = {
                td_calculate_politics_effect = yes
                random_list = {
                    1 = { modifier = { multiply = var:td_chance } change_variable = { name = td_successes add = 1 } }
                    1 = { modifier = { multiply = td_failure_weight } change_variable = { name = td_failures add = 1 } }
                }
                if = { limit = { var:td_successes >= td_bill_successes } td_pass_bill_effect = yes }
                else_if = {
                    limit = { var:td_failures >= td_bill_failures }
                    td_cancel_bill_effect = yes
                    trigger_event = td.13
                }
                else = { trigger_event = { id = td.1 days = td_bill_round_days } }
            }
        }
    }
}

td.2 = {
    type = country_event
    hidden = yes
    immediate = {
        if = {
            limit = { has_variable = td_fast scope:td_project_run = var:td_generation }
            if = {
                limit = { td_admission_target_valid_trigger = no }
                td_end_project_effect = yes
            }
            else = {
                td_refresh_target_effect = yes
                if = {
                    limit = { td_fast_ready_trigger = yes }
                    change_variable = { name = td_valid_months add = 1 }
                    set_variable = { name = td_invalid_months value = 0 }
                }
                else = { change_variable = { name = td_invalid_months add = 1 } }
                if = {
                    limit = { var:td_valid_months >= td_fast_months }
                    var:td_project_target = { set_state_type = incorporated }
                    td_end_project_effect = yes
                    trigger_event = td.15
                }
                else_if = {
                    limit = { var:td_invalid_months >= td_fallback_months }
                    td_end_project_effect = yes
                    td_start_regular_effect = yes
                    trigger_event = td.16
                }
                else = { trigger_event = { id = td.2 months = 1 } }
            }
        }
    }
}

td.21 = {
    type = country_event
    title = td.21.t
    desc = td.21.d
    flavor = td.flavor
    immediate = {
        td_refresh_target_effect = yes
        set_variable = { name = td_preview_reason_bonus value = td_reason_bonus }
        set_variable = { name = td_preview_infamy value = td_claim_infamy }
        set_variable = { name = td_preview_regular_cost value = td_regular_project_cost }
        set_variable = { name = td_preview_fast_cost value = td_fast_project_cost }
    }
    option = {
        name = td.submit_bill
        trigger = {
            td_can_start_bill_trigger = yes td_reason_valid_trigger = yes
            OR = {
                AND = { var:td_pending_bill = 1 td_claim_target_valid_trigger = yes }
                AND = { var:td_pending_bill = 2 td_admission_target_valid_trigger = yes var:td_target = { td_needs_admission_trigger = yes } }

        set_variable = { name = td_bill value = var:td_pending_bill }
        td_start_bill_effect = yes
    }
    option = { name = td.change_reason trigger = { td_can_start_bill_trigger = yes } trigger_event = td.20 ai_chance = { base = 0 } }
    option = { name = td.cancel_menu remove_variable = td_pending_bill ai_chance = { base = 0 } }
}
